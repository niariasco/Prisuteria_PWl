CREATE TABLE productos_personalizados_componentes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    producto_personalizado_id INT NOT NULL,
    criterio_id INT NOT NULL,
    opcion_id INT NOT NULL,
    FOREIGN KEY (producto_personalizado_id) REFERENCES productos_personalizados(id) ON DELETE CASCADE,
    FOREIGN KEY (criterio_id) REFERENCES criterios(id) ON DELETE CASCADE,
    FOREIGN KEY (opcion_id) REFERENCES opciones(id) ON DELETE CASCADE
);

DELIMITER $$

CREATE TRIGGER tr_update_precio_total_after_insert
AFTER INSERT ON productos_personalizados_componentes
FOR EACH ROW
BEGIN
    DECLARE precio_base DECIMAL(10,2);
    DECLARE precio_adicional_total DECIMAL(10,2);

    -- Obtener precio del producto base
    SELECT precio INTO precio_base
    FROM productos
    WHERE productosId = (SELECT producto_base_id 
                         FROM productos_personalizados 
                         WHERE id = NEW.producto_personalizado_id);

    -- Sumar precio de todas las opciones seleccionadas para este producto personalizado
    SELECT IFNULL(SUM(o.precio_adicional),0) INTO precio_adicional_total
    FROM productos_personalizados_componentes ppc
    INNER JOIN opciones o ON ppc.opcion_id = o.id
    WHERE ppc.producto_personalizado_id = NEW.producto_personalizado_id;

    -- Actualizar precio total
    UPDATE productos_personalizados
    SET precio_total = precio_base + precio_adicional_total
    WHERE id = NEW.producto_personalizado_id;
END$$

DELIMITER ;
