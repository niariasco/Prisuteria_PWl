drop table promocion_categorias;
drop table promocion_productos;
DROP TABLE IF EXISTS `promociones`;

CREATE TABLE `promociones` (
  `id` INT NOT NULL AUTO_INCREMENT,
  `nombre` VARCHAR(100) NOT NULL,
  `tipo` ENUM('Producto', 'Categoria') NOT NULL,
  `descuento` DECIMAL(5,2) NOT NULL,
  `fecha_inicio` DATETIME NOT NULL,
  `fecha_fin` DATETIME NOT NULL,
  activo BOOLEAN DEFAULT TRUE,
  `ProductoID` INT DEFAULT NULL,
  `CategoriaID` INT DEFAULT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`ProductoID`) REFERENCES `productos` (`productosId`) ON DELETE SET NULL,
  FOREIGN KEY (`CategoriaID`) REFERENCES `categorias` (`categoriaId`) ON DELETE SET NULL
);
