DELIMITER //

CREATE FUNCTION obtener_precio_final(productoId INT)
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
  DECLARE precio_base DECIMAL(10,2);
  DECLARE descuento_producto DECIMAL(5,2);
  DECLARE descuento_categoria DECIMAL(5,2);
  DECLARE descuento_aplicado DECIMAL(5,2);
  DECLARE categoriaId INT;

  -- Obtener precio base y categoría
  SELECT precio, categoria_id INTO precio_base, categoriaId
  FROM productos
  WHERE productosId = productoId;

  -- Descuento por producto (solo si está activo y en fecha)
  SELECT descuento INTO descuento_producto
  FROM promociones
  WHERE tipo = 'Producto'
    AND ProductoID = productoId
    AND activo = 1
    AND NOW() BETWEEN fecha_inicio AND fecha_fin
  ORDER BY descuento DESC
  LIMIT 1;

  -- Descuento por categoría (si aplica)
  SELECT descuento INTO descuento_categoria
  FROM promociones
  WHERE tipo = 'Categoria'
    AND CategoriaID = categoriaId
    AND activo = 1
    AND NOW() BETWEEN fecha_inicio AND fecha_fin
  ORDER BY descuento DESC
  LIMIT 1;

  -- Elegir el mayor descuento si existen ambos
  SET descuento_aplicado = GREATEST(COALESCE(descuento_producto, 0), COALESCE(descuento_categoria, 0));

  -- Aplicar descuento
  RETURN ROUND(precio_base * (1 - (descuento_aplicado / 100)), 2);
END;
//

DELIMITER ;


DELIMITER //

CREATE TRIGGER tr_precio_detalle_orden
BEFORE INSERT ON detalle_orden
FOR EACH ROW
BEGIN
  DECLARE precio_final DECIMAL(10,2);
  SET precio_final = obtener_precio_final(NEW.producto_id);
  SET NEW.precio_unitario = precio_final;
  SET NEW.subtotal = precio_final * NEW.cantidad;
END;
//

DELIMITER ;


DELIMITER //

CREATE TRIGGER tr_total_orden
AFTER INSERT ON orden_detalle
FOR EACH ROW
BEGIN
  UPDATE ordenes
  SET total = (
    SELECT SUM(subtotal)
    FROM detalle_orden
    WHERE orden_id = NEW.orden_id
  )
  WHERE ordenesId = NEW.orden_id;
END;
//

DELIMITER ;

CREATE TABLE detalle_orden (
    detalleId INT(11) NOT NULL AUTO_INCREMENT,
    orden_id INT(11) NOT NULL,
    producto_id INT(11) NOT NULL,
    cantidad INT(11) NOT NULL,
    precio_unitario DECIMAL(10,2) NOT NULL,
    subtotal DECIMAL(10,2) NOT NULL,
    PRIMARY KEY (detalleId),
    KEY orden_id (orden_id),
    KEY producto_id (producto_id),
    CONSTRAINT detalle_orden_fk_orden FOREIGN KEY (orden_id) REFERENCES ordenes (ordenesId) ON DELETE CASCADE,
    CONSTRAINT detalle_orden_fk_producto FOREIGN KEY (producto_id) REFERENCES productos (productosId)
) 




UPDATE detalle_orden d
JOIN productos p ON d.producto_id = p.productosId
LEFT JOIN promociones pr ON pr.ProductoID = p.productosId
  AND pr.activo = 1
  AND NOW() BETWEEN pr.fecha_inicio AND pr.fecha_fin
SET 
  d.precio_unitario = COALESCE(p.precio * (1 - pr.descuento / 100), p.precio),
  d.subtotal = d.cantidad * COALESCE(p.precio * (1 - pr.descuento / 100), p.precio)
WHERE d.precio_unitario = 0 OR d.subtotal = 0;

UPDATE ordenes o

  o.impuestos = ROUND(dsum.suma_subtotal * 0.13, 2),
  o.total = ROUND(dsum.suma_subtotal * 1.13, 2);
